#define __SFR_OFFSET 0x00
#include "avr/io.h"

.global gpio_init
.global red_on
.global red_off

.section .text

gpio_init:
  SBI DDRD, 2   ; Red
  SBI DDRD, 3   ; Green
  SBI DDRD, 4   ; Yellow
  SBI DDRD, 5   ; Buzzer
  RET

red_on:
  SBI PORTD, 2
  RET

red_off:
  CBI PORTD, 2
  RET
